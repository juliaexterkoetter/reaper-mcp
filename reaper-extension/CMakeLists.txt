cmake_minimum_required(VERSION 3.24)
project(reaper_mcp VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The extension currently targets Windows x64")
endif()
include(FetchContent)
FetchContent_Declare(reaper_sdk GIT_REPOSITORY https://github.com/justinfrankel/reaper-sdk.git GIT_TAG 08e885d14c87c2973ec301d220ff9fe6b6a7cbeb)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG 55f93686c01528224f448c19128836e7df245f72)
FetchContent_MakeAvailable(reaper_sdk json)
add_library(reaper_mcp SHARED src/main.cpp)
target_compile_features(reaper_mcp PRIVATE cxx_std_17)
target_include_directories(reaper_mcp PRIVATE include ${reaper_sdk_SOURCE_DIR}/sdk)
target_link_libraries(reaper_mcp PRIVATE nlohmann_json::nlohmann_json ws2_32)
target_compile_definitions(reaper_mcp PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
set_target_properties(reaper_mcp PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(reaper_mcp PRIVATE /W4 /EHsc /utf-8)
endif()
install(TARGETS reaper_mcp RUNTIME DESTINATION .)
install(FILES ${reaper_sdk_SOURCE_DIR}/sdk/LICENSE DESTINATION licenses RENAME REAPER-SDK.txt)
install(FILES ${json_SOURCE_DIR}/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json.txt)
include(CTest)
if(BUILD_TESTING)
  add_executable(bridge_test tests/bridge_test.cpp)
  target_compile_features(bridge_test PRIVATE cxx_std_17)
  target_include_directories(bridge_test PRIVATE include)
  target_compile_definitions(bridge_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(bridge_test PRIVATE nlohmann_json::nlohmann_json ws2_32)
  add_test(NAME bridge_security COMMAND bridge_test)
endif()
